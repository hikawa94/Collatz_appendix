def Glide_check(number):
      M=""
      first="Y"
      flag=0
      zero=0
      one=1
      n =zero
      parity_vector=""
      N=number
      n2=number
      while True:
          if n2 <= 1 :
              break    ## if ( $n2 <= 1) { break }
          n = n + 1 
          if  (n2 % 2) == 0:   ##if (n2 % 2 == 0)
              n1 = n2
              n2 = n2//2        ##   n2=n2/2
              if ( flag == 0) and (first=="Y") and (n2<N): 
                 flag = 1
                 M = n 
                 R = n2 
                 break
              if (flag == 1) and (first=="Y" ):
                 first="N"
          else:
              n1 = n2
              n2 = n2*3   ## $n2 = $n2 * 3
              n2 = n2+1   ## n2 = n2 * 3 + 1
              n2 = n2//2  ## $n2 = ($n2 * 3 + 1)/2
              if ( flag == 0) and (first=="Y") and (n2<N ):
                  flag = 1
                  M = n 
                  R = n2 
              if (flag == 1) and ( first=="Y" ):
                  first="N"
      out1=M  ## Glide
      out2="not used" ##
      return  out1,out2
