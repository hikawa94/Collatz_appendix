function Glide_check($number){

      $parity_vector="";

      $n2=$number;
  while (1) {
     if( gmp_cmp($n2,$one)<=0 ) { break; }  // if ( $n2 <= 1) { break; }
        $n = $n + 1; 

        if (gmp_div_r($n2,2) == 0) {  //if ($n2 % 2 == 0) {
            $n1 = $n2;
            $n2 = gmp_div_q($n2, 2);   //   $n2=$n2/2;
            if ( $flag == 0 && $first=="Y" && gmp_cmp($n2,$N) < 0  ) {  
                 $flag = 1;
                 $M = $n ;
                 $R = $n2 ;
            }

            if($flag == 1 && $first=="Y" ){
                 $first="N";
            }
        } else {
            $n1 = $n2;
            $n2 = gmp_mul($n2, 3) ;  // $n2 = $n2 * 3;
            $n2 = gmp_add($n2,1);   // $n2 = $n2 * 3 + 1;
            $n2 = gmp_div_q($n2, 2);  // $n2 = ($n2 * 3 + 1)/2
            if ( $flag == 0 && $first=="Y" && gmp_cmp($n2,$N) < 0 ) {  
                 $flag = 1;
                 $M = $n ;
                 $R = $n2 ;
            }
        }
    }
    $out1=$M; // Glide
    $out2="not used"; //
    return  array($out1,$out2);
}
