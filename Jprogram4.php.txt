function Glide_check($number){ 
      $M="";
      $first="Y";
      $flag=0;
      $zero=gmp_init(0);
      $one=gmp_init(1);
      $n =$zero;
      $parity_vector="";

      $N=$number;
      $n2=$number;
  while (1) {
     if( gmp_cmp($n2,$one)<=0 ) { break; }  // if ( $n2 <= 1) { break; }
        $n = $n + 1; 

        if (gmp_div_r($n2,2) == 0) {  //if ($n2 % 2 == 0) {
            $n1 = $n2;
            $n2 = gmp_div_q($n2, 2);   //   $n2=$n2/2;
            $parity_vector=$parity_vector."0";
            if ( $flag == 0 && $first=="Y" && gmp_cmp($n2,$N) < 0  ) {  
                 $flag = 1;
                 $M = $n ;
                 $R = $n2 ;
                 break;

            }

            if($flag == 1 && $first=="Y" ){
                 $first="N";
            }
        } else {
            $n1 = $n2;
            $n2 = gmp_mul($n2, 3) ;  // $n2 = $n2 * 3;
            $n2 = gmp_add($n2,1);   // $n2 = $n2 * 3 + 1;
            $n2 = gmp_div_q($n2, 2);  // $n2 = ($n2 * 3 + 1)/2
            $parity_vector=$parity_vector."1";

           if ( $flag == 0 && $first=="Y" && gmp_cmp($n2,$N) < 0 ) {  
                 $flag = 1;
                 $M = $n ;
                 $R = $n2 ;
            }
            if($flag == 1 && $first=="Y" ){
                $first="N";
            }
        }
    }
    $out1=$M; // Glide
    $out2=$parity_vector; // $parity_vector;
    $out3="not used";
    return  array($out1,$out2,$out3);

}
